short_circuit.sy

  2  int hits = 0;
  3  int touch() { hits = hits + 1; return 1; }
  4  int main() {
  5      int n = getint();
  6      if (n != 0 && 12 / n > 2) putint(1); else putint(0);
  7      putch(10);
  8      if (n == 0 || touch()) putint(2); else putint(3);
  9      putch(10);
 10      putint(hits); putch(10);
 11      return 0;
 12  }

short_circuit.ll（条件及合流）

 14  entry:
 15    %n = call i32 @getint()
 16    %nonzero = icmp ne i32 %n, 0
 17    br i1 %nonzero, label %and.rhs, label %and.false
 18  and.rhs:
 19    %q = sdiv i32 12, %n
 20    %large = icmp sgt i32 %q, 2
 21    br i1 %large, label %and.true, label %and.false
 22  and.true:
 23    br label %and.join
 24  and.false:
 25    br label %and.join
 26  and.join:
 27    %and.value = phi i32 [1, %and.true], [0, %and.false]
 28    call void @putint(i32 %and.value)
 29    call void @putch(i32 10)
